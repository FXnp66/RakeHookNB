local infStaminaEnabled = true

local function ffc(parent, name)
    if not parent then return nil end
    local ok, res = pcall(function()
        return parent:FindFirstChild(name)
    end)
    return ok and res or nil
end

local function ffcr(parent, name)
    if not parent then return nil end
    local ok, res = pcall(function()
        return parent:FindFirstChild(name, true)
    end)
    return ok and res or nil
end

local function valOf(v, def)
    if not v then return def end
    local ok, res = pcall(function()
        return v.Value
    end)
    if ok and res ~= nil then return res end
    return def
end

local function eachGc(fn)
    if type(getgc) ~= "function" then return false end
    local ok, res = pcall(getgc, true)
    if not ok or type(res) ~= "table" then return false end
    for _, v in res do
        pcall(fn, v)
    end
    return true
end

local infSys = {
    tabs = setmetatable({}, { __mode = "k" }),
    token = 0,
    fast = 1,
    scan = 2.5,
}

local function patchInfTab(v)
    if type(v) ~= "table" then return false end

    local hit = false
    if infStaminaEnabled and rawget(v, "STAMINA_REGEN") ~= nil then
        v.STAMINA_REGEN = 100
        v.JUMP_STAMINA = 0
        v.JUMP_COOLDOWN = 0
        v.STAMINA_TAKE = 0
        v.stamina = 100
        hit = true
    end
    return hit
end

local function applyInfTabs(scan)
    if not infStaminaEnabled then return end

    for v in infSys.tabs do
        pcall(patchInfTab, v)
    end

    if scan == true then
        eachGc(patchInfTab)
    end
end

local function queueInfTabs(loops)
    if not infStaminaEnabled then return end

    infSys.token += 1
    local token = infSys.token
    task.spawn(function()
        for i = 1, loops or 12 do
            if token ~= infSys.token then return end
            applyInfTabs(i == 1)
            task.wait(i <= 4 and 0.25 or 0.75)
        end
    end)
end

end

local function fillStamina(ch)
    if type(ch) ~= "table" then return false end
    local vars = rawget(ch, "vars")
    if type(vars) ~= "table" then return false end
    local cfg = rawget(ch, "CONFIG")
    local max = cfgVal(cfg, "MAX_STAMINA", 100)

    pcall(function()
        vars.stamina = max
        vars.can_jump = true
        vars.can_jump2 = true
        vars.lastJump = 0
        vars.handlingSRegen = false
        vars.regeningS = false
        vars.breathing = false
        vars.GT_VTZ = true
        local fov = rawget(ch, "sprintFovV")
        if typeof(fov) == "Instance" and fov:IsA("NumberValue") and fov.Value < 0 then
            fov.Value = 0
        end
    end)
    return true
end

local stamMods = setmetatable({}, { __mode = "k" })
local mainMods = setmetatable({}, { __mode = "k" })
local modRoots = setmetatable({}, { __mode = "k" })

local function isLiveInst(obj)
    if typeof(obj) ~= "Instance" then return false end
    local ok, par = pcall(function() return obj.Parent end)
    if not ok or par == nil then return false end
    local ok2, live = pcall(function() return obj:IsDescendantOf(game) end)
    return ok2 and live == true
end

local function isLiveMod(mod)
    return isLiveInst(mod) and mod:IsA("ModuleScript")
end

local function safeReq(mod)
    if type(require) ~= "function" then return false end
    if not isLiveMod(mod) then return false end
    return pcall(require, mod)
end

local function addEnv(list, seen, env)
    if type(env) == "table" and not seen[env] then
        seen[env] = true
        list[#list + 1] = env
    end
end

local function getModuleEnvList()
    local list = {}
    local seen = {}
    addEnv(list, seen, _G)
    pcall(function() addEnv(list, seen, shared) end)
    pcall(function() addEnv(list, seen, rawget(_G, "shared")) end)

    if type(getfenv) == "function" then
        for i = 0, 6 do
            pcall(function()
                addEnv(list, seen, getfenv(i))
            end)
        end
    end
    if type(getrenv) == "function" then
        pcall(function()
            local env = getrenv()
            addEnv(list, seen, env)
            if type(env) == "table" then
                addEnv(list, seen, rawget(env, "_G"))
                addEnv(list, seen, rawget(env, "shared"))
            end
        end)
    end
    return list
end

local function fetchRootModule()
    local envs = getModuleEnvList()
    for i = 1, #envs do
        local env = envs[i]
        local box = type(env) == "table" and rawget(env, "C09KOPGRE09430989PSD39090W3R") or nil
        local fn = type(box) == "table" and rawget(box, "POKDS908") or nil
        if type(fn) == "function" then
            local ok, mod = pcall(fn, "2390WE890OFD-0F")
            if ok and isLiveMod(mod) then
                modRoots[mod] = true
                return mod
            end
        end
    end
    return nil
end

local function patchMainClientModule(mod)
    if not isLiveMod(mod) then return false end
    if mainMods[mod] then return true end
    local ok, lib = safeReq(mod)
    if not ok or type(lib) ~= "table" then return false end

    local old = {}
    for _, name in { "new", "UpdateHB", "UpdateRS", "Update" } do
        local fn = rawget(lib, name)
        if type(fn) == "function" then
            old[name] = fn
        end
    end
    if not next(old) then return false end

    mainMods[mod] = { lib = lib, old = old }

    for _, name in { "UpdateHB", "UpdateRS", "Update" } do
        local fn = old[name]
        if fn then
            lib[name] = function(ch, ...)
                if infStaminaEnabled then fillStamina(ch) end
                local a, b, c = fn(ch, ...)
                if infStaminaEnabled then fillStamina(ch) end
                return a, b, c
            end
        end
    end
    return true
end

local function patchStaminaModule(mod)
    if not isLiveMod(mod) then return false end
    if stamMods[mod] then return true end
    if mod.Name ~= "M_H" then return false end

    local ok, lib = safeReq(mod)
    if not ok or type(lib) ~= "table" then return false end
    if type(rawget(lib, "TakeStamina")) ~= "function" and type(rawget(lib, "Update")) ~= "function" then
        return false
    end

    local old = {}
    for _, name in { "CanSprintCheck", "JumpHandler", "TakeStamina", "StaminaDrain", "DoRegenStamina", "RegenStaminaHandler", "Update" } do
        local fn = rawget(lib, name)
        if type(fn) == "function" then
            old[name] = fn
        end
    end

    stamMods[mod] = { lib = lib, old = old }

    if old.CanSprintCheck then
        lib.CanSprintCheck = function(ch, ...)
            if infStaminaEnabled then
                fillStamina(ch)
                local vars = type(ch) == "table" and rawget(ch, "vars")
                local hum = type(ch) == "table" and rawget(ch, "hum")
                return type(vars) == "table" and vars.canMove ~= false and vars.canMove2 ~= false and (typeof(hum) ~= "Instance" or hum.Health > 0)
            end
            return old.CanSprintCheck(ch, ...)
        end
    end

    if old.JumpHandler then
        lib.JumpHandler = function(ch, state, ...)
            if infStaminaEnabled and state == false then
                fillStamina(ch)
                state = true
            end
            return old.JumpHandler(ch, state, ...)
        end
    end

    if old.TakeStamina then
        lib.TakeStamina = function(ch, amount, ...)
            if infStaminaEnabled and type(amount) == "number" and amount > 0 then
                fillStamina(ch)
                return
            end
            local a, b, c = old.TakeStamina(ch, amount, ...)
            if infStaminaEnabled then fillStamina(ch) end
            return a, b, c
        end
    end

    if old.StaminaDrain then
        lib.StaminaDrain = function(ch, ...)
            if infStaminaEnabled then
                fillStamina(ch)
                return
            end
            return old.StaminaDrain(ch, ...)
        end
    end

    if old.DoRegenStamina then
        lib.DoRegenStamina = function(ch, ...)
            if infStaminaEnabled then
                fillStamina(ch)
                return
            end
            return old.DoRegenStamina(ch, ...)
        end
    end

    if old.RegenStaminaHandler then
        lib.RegenStaminaHandler = function(ch, ...)
            if infStaminaEnabled then
                fillStamina(ch)
                return
            end
            return old.RegenStaminaHandler(ch, ...)
        end
    end

    if old.Update then
        lib.Update = function(ch, ...)
            if infStaminaEnabled then fillStamina(ch) end
            local a, b, c = old.Update(ch, ...)
            if infStaminaEnabled then fillStamina(ch) end
            return a, b, c
        end
    end
    return true
end

local function restoreStaminaModules()
    for mod, data in stamMods do
        pcall(function()
            local lib = data.lib
            for name, fn in data.old or {} do
                lib[name] = fn
            end
        end)
        stamMods[mod] = nil
    end
end

local function addModuleCandidate(obj, seen, list)
    if isLiveMod(obj) and obj.Name == "M_H" and not seen[obj] then
        seen[obj] = true
        list[#list + 1] = obj
    end
end

local function findStaminaModules()
    local seen = {}
    local list = {}
    local root = fetchRootModule()
    if isLiveMod(root) then
        patchMainClientModule(root)
        for _, obj in root:GetDescendants() do
            addModuleCandidate(obj, seen, list)
        end
    end
    for rootMod in modRoots do
        if isLiveMod(rootMod) then
            patchMainClientModule(rootMod)
            for _, obj in rootMod:GetDescendants() do
                addModuleCandidate(obj, seen, list)
            end
        else
            modRoots[rootMod] = nil
        end
    end
    if type(getloadedmodules) == "function" then
        pcall(function()
            for _, mod in getloadedmodules() do
                addModuleCandidate(mod, seen, list)
            end
        end)
    end
    if type(getnilinstances) == "function" then
        pcall(function()
            for _, obj in getnilinstances() do
                addModuleCandidate(obj, seen, list)
            end
        end)
    end
    for _, root in { player, ffc(player, "PlayerScripts"), ffc(player, "PlayerGui"), ffc(player, "Backpack"), RS } do
        if root then
            for _, obj in root:GetDescendants() do
                addModuleCandidate(obj, seen, list)
            end
        end
    end
    return list
end

local function applyStaminaModuleBypass()
    if not infStaminaEnabled then return 0 end
    local n = 0
    local root = fetchRootModule()
    if isLiveMod(root) and patchMainClientModule(root) then
        n += 1
    end
    for _, mod in findStaminaModules() do
        if patchStaminaModule(mod) then
            n += 1
        end
    end
    return n
end

local gstmnaFn = function() return 100 end
local gstmnaOwned = false
local gstmnaInst = nil
local stamTk = setmetatable({}, { __mode = "k" })

local function restoreStaminaSignals()
    for c in stamTk do
        pcall(function()
            if type(c.Enable) == "function" then
                c:Enable()
            elseif c.Enabled ~= nil then
                c.Enabled = true
            end
        end)
        stamTk[c] = nil
    end
    if gstmnaOwned and gstmnaInst and gstmnaInst.Parent and gstmnaInst:IsA("BindableFunction") then
        pcall(function() gstmnaInst.OnInvoke = nil end)
    end
    gstmnaOwned = false
    gstmnaInst = nil
end

local function applyStaminaSignals()
    local gst = ffc(RS, "GSTMNA")
    if gst and gst:IsA("BindableFunction") then
        if infStaminaEnabled then
            pcall(function()
                gst.OnInvoke = gstmnaFn
                gstmnaOwned = true
                gstmnaInst = gst
            end)
        elseif gstmnaOwned and gstmnaInst == gst then
            pcall(function() gst.OnInvoke = nil end)
            gstmnaOwned = false
            gstmnaInst = nil
        end
    end

    local tk = ffc(RS, "TKSMNA")
    if not (tk and tk:IsA("BindableEvent") and type(getconnections) == "function") then
        if not infStaminaEnabled then
            restoreStaminaSignals()
        end
        return false
    end

    local ok, cons = pcall(getconnections, tk.Event)
    if not ok or type(cons) ~= "table" then return false end

    for _, c in cons do
        if infStaminaEnabled then
            pcall(function()
                if type(c.Disable) == "function" then
                    c:Disable()
                    stamTk[c] = true
                elseif c.Enabled ~= nil then
                    c.Enabled = false
                    stamTk[c] = true
                end
            end)
        elseif stamTk[c] then
            pcall(function()
                if type(c.Enable) == "function" then
                    c:Enable()
                elseif c.Enabled ~= nil then
                    c.Enabled = true
                end
            end)
            stamTk[c] = nil
        end
    end
    return true
end

local function setupSpeedForCharacter(char)

    for _, conn in ipairs(speedConnections) do
        pcall(function() conn:Disconnect() end)
    end
    speedConnections = {}

    local hum = char:WaitForChild("Humanoid", 5)
    if not hum then return end

    applySpeed(hum)

    table.insert(speedConnections, hum:GetPropertyChangedSignal("WalkSpeed"):Connect(function()
        if hum.WalkSpeed ~= playerSpeed then

        end

    table.insert(speedConnections, RunService.Heartbeat:Connect(function()
        pcall(function()
            if hum and hum.Parent then
                applySpeed(hum)
            end
        end)
    end))
end

local function fmtTime(v)
    v = math.max(0, math.floor(tonumber(v) or 0))
    local h = math.floor(v / 3600)
    local m = math.floor((v % 3600) / 60)
    local s = v % 60
    if h > 0 then
        return string.format("%d:%02d:%02d", h, m, s)
    end
    return string.format("%d:%02d", m, s)
end

local function fmtPower(v)
    local pct = math.max(0, (tonumber(v) or 1000) / 10)
    if pct % 1 == 0 then
        return tostring(math.floor(pct))
    end
    return string.format("%.1f", pct)
end

local function infoText()
    return infoTgt .. "\n" .. infoTime .. "\n" .. infoPower
end

local function getRakeInfo(rake)
    if not rake then return "?", "?" end
    local hum = rake:FindFirstChildOfClass("Humanoid")
    local health = "?"
    if hum then
        health = math.floor(hum.Health) .. "/" .. math.floor(hum.MaxHealth)
    end

    local char = player.Character
    local dist = "?"
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local rakeHrp = rake:FindFirstChild("HumanoidRootPart") or rake.PrimaryPart or rake:FindFirstChildWhichIsA("BasePart")
        if hrp and rakeHrp then
            dist = math.floor((hrp.Position - rakeHrp.Position).Magnitude) .. "m"
        end
    end

    return health, dist
end

local function doTeleport()
    local char = player.Character
    if not char then
        Library:Notify({
            Title = "传送",
            Description = "角色未加载",
            Time = 2,
        })
        return
    end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then
        Library:Notify({
            Title = "传送",
            Description = "未找到 HumanoidRootPart",
            Time = 2,
        })
        return
    end
    local rake = Workspace:FindFirstChild("Rake") or Workspace:FindFirstChild("RakoofNPC")
    if rake then
        local rakeHRP = rake:FindFirstChild("HumanoidRootPart") or rake:FindFirstChild("Head")
        if rakeHRP then

            local dist = (hrp.Position - rakeHRP.Position).Magnitude
            if dist > 30 then
                Library:Notify({
                    Title = "传送",
                    Description = "Rake 距离过远 (" .. string.format("%.1f", dist) .. "m > 30m)",
                    Time = 2,
                })
                return
            end
            local behindOffset = rakeHRP.CFrame.LookVector * -8
            local targetPos = rakeHRP.Position + behindOffset + Vector3.new(0, 0.5, 0)
            hrp.CFrame = CFrame.new(targetPos, rakeHRP.Position)
            Library:Notify({
                Title = "传送",
                Description = "已传送到 Rake 身后 (距离: " .. string.format("%.1f", dist) .. "m)",
                Time = 2,
            })
        else
            Library:Notify({
                Title = "传送",
                Description = "未找到 Rake 的 HumanoidRootPart",
                Time = 2,
            })
        end
    else
        Library:Notify({
            Title = "传送",
            Description = "未找到 Rake",
            Time = 2,
        })
    end
end

local trapClickConn = nil
local trapDragMainLoop = nil

local function td_getTrapsFolder()
    local debris = Workspace:FindFirstChild("Debris")
    if debris then
        local traps = debris:FindFirstChild("Traps")
        if traps then return traps end
    end
    local map = Workspace:FindFirstChild("Map")
    if map then
        local traps = map:FindFirstChild("Traps")
        if traps then return traps end
    end
    return Workspace:FindFirstChild("Traps")
end

local function td_syncTrapPosition(trap, cf)
    if not trap or not trap.Parent then return end
    pcall(function()

        local trpEvent = RS:FindFirstChild("TRP_Event")
        if trpEvent and trpEvent:IsA("RemoteEvent") then
            trpEvent:FireServer(true)
            trpEvent:FireServer(trap, cf.Position)
        end
    end)
    pcall(function()
        local remoteNames = {
            "UpdateTrapPosition", "TrapUpdate", "SyncTrap", "TrapSync",
            "MoveTrap", "TrapMove", "SetTrapPosition", "TrapSetPosition",
            "UpdateTrap", "ReplicateTrap", "TrapReplicate"
        }
        for _, name in ipairs(remoteNames) do
            local remote = RS:FindFirstChild(name)
            if remote and remote:IsA("RemoteEvent") then
                remote:FireServer(trap, cf)
                break
            end
        end
    end)
end

local function td_fastTrap(trap)
    if not trap or not trap.Parent then return end
    pcall(function()
        local hitBox = trap:FindFirstChild("HitBox")
        if hitBox and hitBox:IsA("BasePart") then
            hitBox.CanCollide = true
            hitBox.CanQuery = true
            hitBox.CanTouch = true
        end
        for _, part in ipairs(trap:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = true
                part.CanQuery = true
                part.CanTouch = true
            end
        end
        if type(fireproximityprompt) == "function" then
            local att = hitBox and hitBox:FindFirstChildWhichIsA("Attachment", true)
            local prompt = att and att:FindFirstChildWhichIsA("ProximityPrompt")
            if prompt then
                fireproximityprompt(prompt)
            end
        end
    end)
end

local function prepareTrapForDrag(trap)
    if not trap or not trap.Parent then return false end

    local primary = trap.PrimaryPart
    if not primary then
        for _, part in ipairs(trap:GetDescendants()) do
            if part:IsA("BasePart") then
                trap.PrimaryPart = part
                primary = part
                break
            end
        end
    end
    if not primary then return false end

    pcall(function()
        for _, part in ipairs(trap:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Anchored = false
                part.CanCollide = true
                part.CanQuery = true
                part.CanTouch = true
                part.Massless = false
            end
        end
    end)

    pcall(function()
        if primary:IsA("BasePart") and primary.CanSetNetworkOwnership then
            primary:SetNetworkOwner(player)
        end
    end)

    return true
end

local function anchorTrap(trap)
    if not trap or not trap.Parent then return end
    pcall(function()
        for _, part in ipairs(trap:GetDescendants()) do
            if part:IsA("BasePart") then
                part.Anchored = true
            end
        end
    end)
end

local function selectNearestTrap()
    local char = player.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end

    local trapsRoot = td_getTrapsFolder()
    if not trapsRoot then return nil end

    local nearestTrap = nil
    local nearestDist = 30

    for _, trap in ipairs(trapsRoot:GetChildren()) do
        if trap:IsA("Model") then

            local isTrap = trap.Name == "RakeTrapModel" or trap.Name:lower():find("rusty") or trap.Name:lower():find("trap")
            if isTrap then
                local trapPart = trap.PrimaryPart or trap:FindFirstChildWhichIsA("BasePart")
                if trapPart then
                    local dist = (hrp.Position - trapPart.Position).Magnitude
                    if dist < nearestDist then
                        nearestDist = dist
                        nearestTrap = trap
                    end
                end
            end
        end
    end

    return nearestTrap
end

local function selectTrap(trap)
    if not trap then return false end
    selectedTrap = trap

    if not selectedTrap.PrimaryPart then
        local anyPart = selectedTrap:FindFirstChildWhichIsA("BasePart")
        if anyPart then selectedTrap.PrimaryPart = anyPart end
    end

    prepareTrapForDrag(selectedTrap)

    pcall(function()
        local char = player.Character
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                local targetPos = hrp.Position + hrp.CFrame.LookVector * 6 + Vector3.new(0, 3, 0)
                selectedTrap:MoveTo(targetPos)
                td_syncTrapPosition(selectedTrap, CFrame.new(targetPos))
            end
        end
    end)

    td_fastTrap(selectedTrap)
    Library:Notify({Title = "陷阱", Description = "已选中: " .. selectedTrap.Name, Time = 2})
    return true
end

local function dropTrapToGround(trap)
    pcall(function()
        if not trap or not trap.Parent then return end

        local char = player.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local dropPos = hrp.Position + hrp.CFrame.LookVector * 5

        local rp = RaycastParams.new()
        rp.FilterType = Enum.RaycastFilterType.Blacklist
        rp.FilterDescendantsInstances = {char, trap}
        local result = Workspace:Raycast(dropPos + Vector3.new(0, 50, 0), Vector3.new(0, -100, 0), rp)
        local groundY = result and result.Position.Y or dropPos.Y

        local undergroundPos = Vector3.new(dropPos.X, groundY - 3, dropPos.Z)

        trap:MoveTo(undergroundPos)

        td_syncTrapPosition(trap, CFrame.new(undergroundPos))

        anchorTrap(trap)

        td_fastTrap(trap)
        pcall(function()
            if type(fireproximityprompt) == "function" then
                local hit = trap:FindFirstChild("HitBox", true)
                local att = hit and hit:FindFirstChildWhichIsA("Attachment", true)
                local prompt = att and att:FindFirstChildWhichIsA("ProximityPrompt")
                if prompt then fireproximityprompt(prompt) end
            end
        end)

        task.delay(0.15, function()
            pcall(function()
                if trap and trap.Parent then
                    trap:MoveTo(undergroundPos)
                    td_syncTrapPosition(trap, CFrame.new(undergroundPos))
                    anchorTrap(trap)
                    td_fastTrap(trap)
                end
            end)
        end)
    end)
end

local function setupTrapClick()
    if trapClickConn then
        pcall(function() trapClickConn:Disconnect() end)
        trapClickConn = nil
    end

    trapClickConn = UIS.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed then return end
        if input.UserInputType ~= Enum.UserInputType.MouseButton1 and input.UserInputType ~= Enum.UserInputType.Touch then return end
        if not dragging then return end

        local nearestTrap = selectNearestTrap()
        if nearestTrap then
            selectTrap(nearestTrap)
        end
    end)
end

local function startTrapDragMainLoop()
    if trapDragMainLoop then return end

    trapDragMainLoop = RunService.RenderStepped:Connect(function()

        if not dragging and selectedTrap then
            local trapToDrop = selectedTrap
            selectedTrap = nil
            dragging = false
            raiseTrap = false
            lowerTrap = false

            dropTrapToGround(trapToDrop)

            Library:Notify({Title = "陷阱", Description = "陷阱已放置到地下", Time = 2})
            return
        end

        if not dragging or not selectedTrap or not selectedTrap.Parent then
            return
        end

        local char = player.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        prepareTrapForDrag(selectedTrap)

        local targetPos = hrp.Position + hrp.CFrame.LookVector * 6
        if raiseTrap then
            targetPos = targetPos + Vector3.new(0, 0.5, 0)
        elseif lowerTrap then
            targetPos = targetPos - Vector3.new(0, 0.5, 0)
        end

        pcall(function()
            selectedTrap:MoveTo(targetPos)
        end)

        td_syncTrapPosition(selectedTrap, CFrame.new(targetPos))

        pcall(function()
            for _, part in ipairs(selectedTrap:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.Anchored = false
                    part.CanCollide = true
                    part.CanQuery = true
                    part.CanTouch = true
                end
            end
        end)

        local trapPart = selectedTrap.PrimaryPart or selectedTrap:FindFirstChildWhichIsA("BasePart")
        if trapPart and (hrp.Position - trapPart.Position).Magnitude > 100 then
            local trapToDrop = selectedTrap
            selectedTrap = nil
            dragging = false
            raiseTrap = false
            lowerTrap = false
            dropTrapToGround(trapToDrop)
            Library:Notify({Title = "陷阱", Description = "距离过远，陷阱已自动放置", Time = 2})
        end
    end)
end

local TrapKillEnabled = false
local TrapKillConnections = {}

local function ClearTrapKillConnections()
    for _, conn in ipairs(TrapKillConnections) do
        pcall(function() conn:Disconnect() end)
    end
    TrapKillConnections = {}
end

local function FindRakeForTrapKill()
    local rake = Workspace:FindFirstChild("Rake")
    if rake then return rake end
    rake = Workspace:FindFirstChild("RakoofNPC")
    if rake then return rake end
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("Model") then
            if v.Name:lower():find("rake") or v.Name:lower():find("rakoof") then
                return v
            end
            local h = v:FindFirstChildOfClass("Humanoid")
            if h and h.MaxHealth == 400 then
                return v
            end
        end
    end
    return nil
end

local function KillRakeInstant()
    if not TrapKillEnabled then return end
    pcall(function()
        local rake = FindRakeForTrapKill()
        if not rake then return end
        local humanoid = rake:FindFirstChildOfClass("Humanoid")
        if humanoid then
            humanoid.Health = 0
        end
    end)
end

local function MonitorSingleTrap(trap)
    if not trap or not trap:IsA("Model") then return end

    local conn1 = trap.DescendantAdded:Connect(function(desc)
        if not TrapKillEnabled then return end
        if desc:IsA("Sound") or desc:IsA("ParticleEmitter") or desc:IsA("BasePart") then
            task.delay(0.05, KillRakeInstant)
        end
    end)
    table.insert(TrapKillConnections, conn1)

    for _, part in ipairs(trap:GetDescendants()) do
        if part:IsA("BasePart") then
            local lastCF = part.CFrame
            local conn2 = RunService.Heartbeat:Connect(function()
                if not TrapKillEnabled then return end
                if not part or not part.Parent then return end
                local currentCF = part.CFrame
                if (currentCF.Position - lastCF.Position).Magnitude > 0.5 then
                    KillRakeInstant()
                end
                lastCF = currentCF
            end)
            table.insert(TrapKillConnections, conn2)
        end
    end
end

local function MonitorTrapsForKill()
    local debris = Workspace:FindFirstChild("Debris")
    if not debris then return end
    local trapsFolder = debris:FindFirstChild("Traps")
    if not trapsFolder then return end

    local conn = trapsFolder.ChildAdded:Connect(function(trap)
        if not TrapKillEnabled then return end
        if trap.Name == "RakeTrapModel" or trap:IsA("Model") then
            task.wait(0.3)
            MonitorSingleTrap(trap)
        end
    end)
    table.insert(TrapKillConnections, conn)

    for _, trap in ipairs(trapsFolder:GetChildren()) do
        MonitorSingleTrap(trap)
    end
end

local function ConnectRakeStun()
    local rake = FindRakeForTrapKill()
    if not rake then return end
    local stun = rake:FindFirstChild("Stun")
    if stun and stun:IsA("BoolValue") then
        local conn = stun:GetPropertyChangedSignal("Value"):Connect(function()
            if not TrapKillEnabled then return end
            if stun.Value == true then
                KillRakeInstant()
            end
        end)
        table.insert(TrapKillConnections, conn)
    end
end

local function CheckRakeNearTrap()
    if not TrapKillEnabled then return end
    pcall(function()
        local rake = FindRakeForTrapKill()
        if not rake then return end
        local rakeHrp = rake:FindFirstChild("HumanoidRootPart") or rake.PrimaryPart
        if not rakeHrp then return end
        local debris = Workspace:FindFirstChild("Debris")
        if not debris then return end
        local traps = debris:FindFirstChild("Traps")
        if not traps then return end
        for _, trap in ipairs(traps:GetChildren()) do
            if trap:IsA("Model") then
                local trapPart = trap.PrimaryPart or trap:FindFirstChildWhichIsA("BasePart")
                if trapPart then
                    local dist = (rakeHrp.Position - trapPart.Position).Magnitude
                    if dist <= 10 then
                        KillRakeInstant()
                    end
                end
            end
        end
    end)
end

local LastRakeHealth = 400
local function CheckRakeHealthForKill()
    if not TrapKillEnabled then return end
    pcall(function()
        local rake = FindRakeForTrapKill()
        if not rake then
            LastRakeHealth = 400
            return
        end
        local humanoid = rake:FindFirstChildOfClass("Humanoid")
        if not humanoid then return end
        local current = humanoid.Health
        if current < LastRakeHealth and current > 0 then
            KillRakeInstant()
        end
        LastRakeHealth = current
    end)
end

local TrapKillLoopRunning = false
local function StartTrapKillLoop()
    if TrapKillLoopRunning then return end
    TrapKillLoopRunning = true
    task.spawn(function()
        while TrapKillLoopRunning do
            if TrapKillEnabled then
                ClearTrapKillConnections()
                MonitorTrapsForKill()
                ConnectRakeStun()
                CheckRakeNearTrap()
                CheckRakeHealthForKill()
            end
            task.wait(0.5)
        end
    end)
end

local function StopTrapKillLoop()
    TrapKillLoopRunning = false
    ClearTrapKillConnections()
end

local fastDropEnabled = false

local function fd_ffc(parent, name)
    if not parent then return nil end
    local ok, res = pcall(function()
        return parent:FindFirstChild(name)
    end)
    return ok and res or nil
end

local function fd_ffca(parent, className)
    if not parent then return nil end
    local ok, res = pcall(function()
        return parent:FindFirstChildWhichIsA(className, true)
    end)
    return ok and res or nil
end

local function fd_fastPrompt(prompt, attrs)
    if not prompt then
        return false
    end
    local ok = pcall(function()
        if not prompt:IsA("ProximityPrompt") then
            error("bad prompt")
        end
        prompt.Enabled = true
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        prompt.ClickablePrompt = true
        prompt.MaxActivationDistance = 50
        prompt.Exclusivity = Enum.ProximityPromptExclusivity.AlwaysShow
    end)
    if ok and attrs == true then
        pcall(function()
            prompt:SetAttribute("Busy", false)
            prompt:SetAttribute("Busy2", false)
            prompt:SetAttribute("Unavailable", false)
            prompt:SetAttribute("Unavailable2", false)
        end)
    end
    return ok
end

local function fd_dropRoot()
    local debris = fd_ffc(Workspace, "Debris")
    if not debris then return nil end
    return fd_ffc(debris, "SupplyCrates") or fd_ffc(debris, "SupplyCreates")
end

local function fd_isDropBox(v)
    return v and v.Name == "Box" and v:IsA("Model")
end

local function fd_fastDrop(box)
    if not box or box.Name ~= "Box" or not box:IsA("Model") then return end

    local guiPart = fd_ffc(box, "GUIPart")
    if guiPart then
        local prompt = fd_ffca(guiPart, "ProximityPrompt")
        if prompt then
            fd_fastPrompt(prompt, true)
            pcall(function()
                if type(fireproximityprompt) == "function" then
                    fireproximityprompt(prompt)
                end
            end)
        end
    end

    local boxPrompt = fd_ffca(box, "ProximityPrompt")
    if boxPrompt then
        fd_fastPrompt(boxPrompt, true)
        pcall(function()
            if type(fireproximityprompt) == "function" then
                fireproximityprompt(boxPrompt)
            end
        end)
    end

    local unlock = fd_ffc(box, "UnlockValue")
    if unlock then
        pcall(function() unlock.Value = 100 end)
    end

    local hitBox = fd_ffc(box, "HitBox")
    if hitBox and hitBox:IsA("BasePart") then
        pcall(function()
            hitBox.CanCollide = false
            hitBox.CanQuery = false
        end)
    end

    for _, child in ipairs(box:GetDescendants()) do
        if child.Name:lower():find("progress") or child.Name:lower():find("unlock") or child.Name:lower():find("open") then
            if child:IsA("IntValue") or child:IsA("NumberValue") then
                pcall(function() child.Value = child.Value + 100 end)
            end
        end
        if child.Name:lower():find("busy") or child.Name:lower():find("cooldown") then
            if child:IsA("BoolValue") then
                pcall(function() child.Value = false end)
            end
        end
    end

    pcall(function()
        local ev = RS:FindFirstChild("SupplyClientEvent")
        if ev then
            ev:FireServer("Open", true)
        end
    end)
end

local function fd_eachDrop(fn)
    local root = fd_dropRoot()
    if not root or type(fn) ~= "function" then
        return 0
    end
    local n = 0
    local hit = {}
    local function run(box)
        if box and not hit[box] and fd_isDropBox(box) then
            hit[box] = true
            n += 1
            pcall(fn, box)
        end
    end
    run(fd_ffc(root, "Box"))
    for _, box in ipairs(root:GetChildren()) do
        run(box)
    end
    if n <= 0 then
        for _, box in ipairs(root:GetDescendants()) do
            run(box)
        end
    end
    return n
end

local function fd_processAllDrops()
    if not fastDropEnabled then return 0 end
    return fd_eachDrop(fd_fastDrop)
end

Workspace.DescendantAdded:Connect(function(obj)
    if fastDropEnabled and obj.Name == "Box" and obj:IsA("Model") then
        task.defer(function()
            fd_fastDrop(obj)
        end)
    end
end)

    if fd_promptTimer >= 0.2 then
        fd_promptTimer = 0

    end
end)
